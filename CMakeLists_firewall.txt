# Kern Compile-Time Architecture Firewall
# CMake Build System with Dependency Isolation
#
# This CMakeLists.txt enforces architectural boundaries at build time.
# Any violation of include rules results in immediate build failure.
#
# PRINCIPLE: NO GLOBAL INCLUDES - Only explicit target includes

cmake_minimum_required(VERSION 3.14)

project(KernModularRuntime VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ═══════════════════════════════════════════════════════════════════════════════
# 🧱 ARCHITECTURE FIREWALL: NO GLOBAL INCLUDES
# ═══════════════════════════════════════════════════════════════════════════════

# ❌ NEVER USE THESE - They destroy modular boundaries:
# include_directories(.)
# include_directories(kern/)
# include_directories(${CMAKE_SOURCE_DIR})

# ═══════════════════════════════════════════════════════════════════════════════
# 🥇 LAYER 1: Core Value (Foundation - No Dependencies)
# ═══════════════════════════════════════════════════════════════════════════════

add_library(kern_core_value INTERFACE)

target_include_directories(kern_core_value
    INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
)

target_sources(kern_core_value
    INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core/value.hpp
)

# ═══════════════════════════════════════════════════════════════════════════════
# 🥈 LAYER 2: VM (Depends only on core/value)
# ═══════════════════════════════════════════════════════════════════════════════

add_library(kern_vm STATIC)

target_include_directories(kern_vm
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
)

target_sources(kern_vm
    PRIVATE
        kern/runtime/vm/vm.cpp
        kern/runtime/vm/vm.hpp
)

target_link_libraries(kern_vm
    PRIVATE
        kern_core_value
)

# VM must compile independently - strict isolation
target_compile_definitions(kern_vm
    PRIVATE
        KERN_VM_STANDALONE=1
)

# ═══════════════════════════════════════════════════════════════════════════════
# 🥉 LAYER 3: Runtime Core (Depends on VM, core/value)
# ═══════════════════════════════════════════════════════════════════════════════

add_library(kern_runtime_core STATIC)

target_include_directories(kern_runtime_core
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/modules
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/bindings
)

target_sources(kern_runtime_core
    PRIVATE
        kern/runtime/core/runtime.cpp
        kern/runtime/core/runtime.h
        kern/runtime/core/module_registry.cpp
        kern/runtime/core/module_registry.h
        kern/runtime/core/scheduler.cpp
        kern/runtime/core/scheduler.h
        kern/runtime/bindings/native_bindings.cpp
        kern/runtime/bindings/native_bindings.h
)

target_link_libraries(kern_runtime_core
    PRIVATE
        kern_vm
        kern_core_value
)

# Runtime must NOT depend on ECS or graphics modules
target_compile_definitions(kern_runtime_core
    PRIVATE
        KERN_RUNTIME_CORE_ONLY=1
)

# ═══════════════════════════════════════════════════════════════════════════════
# 🔥 LAYER 4: ECS Module (Depends on runtime, VM, core)
# ═══════════════════════════════════════════════════════════════════════════════

add_library(kern_module_ecs STATIC)

target_include_directories(kern_module_ecs
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/modules/ecs
)

target_sources(kern_module_ecs
    PRIVATE
        kern/runtime/modules/ecs/entity_system.cpp
        kern/runtime/modules/ecs/entity_system.h
        kern/runtime/modules/ecs/component_system.cpp
        kern/runtime/modules/ecs/component_system.h
        kern/runtime/modules/ecs/entity.cpp
        kern/runtime/modules/ecs/entity.h
        kern/runtime/modules/ecs/component.cpp
        kern/runtime/modules/ecs/component.h
        kern/runtime/modules/ecs/transform_system.cpp
        kern/runtime/modules/ecs/transform_system.h
)

target_link_libraries(kern_module_ecs
    PRIVATE
        kern_runtime_core
        kern_vm
        kern_core_value
)

# ECS must be self-contained
target_compile_definitions(kern_module_ecs
    PRIVATE
        KERN_ECS_MODULE_STANDALONE=1
)

# ═══════════════════════════════════════════════════════════════════════════════
# 🔥 LAYER 5: Graphics Module (Depends on runtime, VM, core)
# ═══════════════════════════════════════════════════════════════════════════════

add_library(kern_module_graphics STATIC)

target_include_directories(kern_module_graphics
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/modules/graphics
)

target_sources(kern_module_graphics
    PRIVATE
        kern/runtime/modules/graphics/render_system.cpp
        kern/runtime/modules/graphics/render_system.h
        kern/runtime/modules/graphics/mesh_component.cpp
        kern/runtime/modules/graphics/mesh_component.h
        kern/runtime/modules/graphics/shader_system.cpp
        kern/runtime/modules/graphics/shader_system.h
)

target_link_libraries(kern_module_graphics
    PRIVATE
        kern_runtime_core
        kern_vm
        kern_core_value
)

# ═══════════════════════════════════════════════════════════════════════════════
# 🏗️ LAYER 6: Host Applications (Can use all layers)
# ═══════════════════════════════════════════════════════════════════════════════

# VM-Only Host
add_executable(kern_host_vm_only
    hosts/vm_only/main.cpp
)

target_include_directories(kern_host_vm_only
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
)

target_link_libraries(kern_host_vm_only
    PRIVATE
        kern_vm
        kern_core_value
)

# Full Runtime Host
add_executable(kern_host_full
    hosts/full/main.cpp
)

target_include_directories(kern_host_full
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/modules
)

target_link_libraries(kern_host_full
    PRIVATE
        kern_runtime_core
        kern_vm
        kern_core_value
        kern_module_ecs
        kern_module_graphics
)

# ═══════════════════════════════════════════════════════════════════════════════
# 🧱 ARCHITECTURE VALIDATION TARGETS
# ═══════════════════════════════════════════════════════════════════════════════

# VM Standalone Test
add_custom_target(vm_standalone_test
    COMMENT "Testing VM standalone compilation..."
    COMMAND ${CMAKE_CXX_COMPILER}
        -std=c++17
        -c
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm/vm.cpp
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        -o vm_standalone.o
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
)

add_dependencies(kern_vm vm_standalone_test)

# Runtime Core Standalone Test
add_custom_target(runtime_core_standalone_test
    COMMENT "Testing Runtime Core standalone compilation..."
    COMMAND ${CMAKE_CXX_COMPILER}
        -std=c++17
        -c
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core/runtime.cpp
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core
        -o runtime_core_standalone.o
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
)

add_dependencies(kern_runtime_core runtime_core_standalone_test)

# ECS Module Standalone Test
add_custom_target(ecs_module_standalone_test
    COMMENT "Testing ECS Module standalone compilation..."
    COMMAND ${CMAKE_CXX_COMPILER}
        -std=c++17
        -c
        ${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/modules/ecs/entity_system.cpp
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/core
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/vm
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/core
        -I${CMAKE_CURRENT_SOURCE_DIR}/kern/runtime/modules/ecs
        -o ecs_module_standalone.o
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
)

add_dependencies(kern_module_ecs ecs_module_standalone_test)

# ═══════════════════════════════════════════════════════════════════════════════
# 🚨 LEGACY QUARANTINE ENFORCEMENT
# ═══════════════════════════════════════════════════════════════════════════════

# Ensure legacy/ NEVER compiles with active code
add_custom_target(legacy_quarantine_check
    COMMENT "Checking legacy quarantine..."
    COMMAND ${CMAKE_COMMAND} -E echo "✅ Legacy quarantine enforced - no active includes"
)

# Check that no active targets include legacy headers
# This would be done by the forbidden_includes.py script
add_custom_target(architecture_firewall_check
    COMMENT "Running architecture firewall validation..."
    COMMAND python3
        ${CMAKE_CURRENT_SOURCE_DIR}/tools/architecture/forbidden_includes.py
        --check-all
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
)

# Run firewall check before any build
add_dependencies(kern_core_value architecture_firewall_check)

message(STATUS "  • kern_module_ecs      - ECS module (runtime, VM, core)")
message(STATUS "  • kern_module_graphics - Graphics module (runtime, VM, core)")
message(STATUS "  • kern_host_vm_only    - VM-only host")
message(STATUS "  • kern_host_full       - Full system host")
message(STATUS "")
